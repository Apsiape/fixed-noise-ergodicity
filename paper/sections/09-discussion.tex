\section{Open problems}\label{sec:discussion}
% =====================================================================

\paragraph{The dimension.} Theorem~A holds on $\Z^{2+k}$ with $k=\max(2,2j)$, and the $j$ of Theorem~C is fixed but very large. Its size comes from the parser of \cref{sec:scaffold}, whose readout uses $\binom{276}{2}+1$ observation points; part~(ii) of Theorem~B needs far fewer, and the computation layer adds only a few to those observation points. Two routes to a small dimension are open. One is a family with small token width: width~$2$ would place Theorem~A on $\Z^6$. The other is a forgetting mechanism whose number of extra dimensions does not grow with~$j$.

\begin{problem}\label{prob:dimension}
What is the least dimension $d$ for which ergodicity of positive-rate cellular automata on $\Z^d$, at one fixed noise rate, is undecidable? What is the least token width of a computable family of Wang sets with undecidable tiling problem?
\end{problem}

\paragraph{The exact degree.} For rational positive rates, ergodicity is a $\Pi^0_2$ property of the table \cite[Remark on the $\Pi^0_2$ placement]{DM26}. Theorem~A shows it is $\Sigma^0_1$-hard at one fixed rate.

\begin{problem}\label{prob:degree}
Is ergodicity of positive-rate cellular automata at one fixed noise rate $\Pi^0_2$-complete?
\end{problem}

\paragraph{Natural rules.} The rules of Theorem~A are built to order. The quorum rule with $k=2$ on $\Z^4$ is the smallest member of its kind, and part~(i) of Theorem~B holds for it.

\begin{problem}\label{prob:natural}
Is there a fixed rule shape in low dimension, applied to arbitrary Wang sets, whose ergodicity at a fixed noise rate is undecidable?
\end{problem}
